import streamlit as st
from groq import Groq

# -------------------------------------
# Streamlit Config
# -------------------------------------

st.set_page_config(
    page_title="ML Learning Assistant",
    page_icon="📊",
    layout="wide"
)

# -------------------------------------
# Groq Client
# -------------------------------------

client = Groq(
    api_key=st.secrets["GROQ_API_KEY"]
)

MODEL = "llama-3.3-70b-versatile"

# -------------------------------------
# Helper Function
# -------------------------------------

def ask_groq(prompt):

    response = client.chat.completions.create(
        model=MODEL,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ],
        temperature=0.3,
        max_tokens=1500
    )

    return response.choices[0].message.content

# -------------------------------------
# UI
# -------------------------------------

st.title("📊 ML Learning Assistant")

st.markdown(
"""
Learn Machine Learning concepts interactively
using Groq LLM.
"""
)

menu = st.sidebar.selectbox(
    "Choose Topic",
    [
        "Supervised Learning",
        "Unsupervised Learning",
        "ML Lifecycle",
        "Evaluation Metrics",
        "ML Quiz",
        "ML Tutor"
    ]
)

# =====================================
# SUPERVISED LEARNING
# =====================================

if menu == "Supervised Learning":

    st.header("📘 Supervised Learning")

    question = st.text_input(
        "Ask About Supervised Learning",
        "What is supervised learning?"
    )

    if st.button("Explain"):

        prompt = f"""
        Explain:

        {question}

        Include:
        - Definition
        - Examples
        - Algorithms
        - Advantages
        - Use Cases

        Explain for beginners.
        """

        with st.spinner("Thinking..."):
            st.markdown(ask_groq(prompt))

# =====================================
# UNSUPERVISED LEARNING

        prompt = f"""
        Explain:

        {question}

        Include:
        - Definition
        - Clustering

        - Real world examples
        """

        with st.spinner("Thinking..."):
            st.markdown(ask_groq(prompt))

# ML LIFECYCLE
